checkout() {
    local source=$1 revision=$2 dest=$3
    if [[ -e "$dest/.git" ]]; then
        if [[ $(git -C "$dest" rev-parse HEAD) != $(git -C "$source" rev-parse "$revision^{commit}") ]]; then
            printf 'Existing checkout has a different revision: %s\n' "$dest" >&2
            return 1
        fi
    else
        git clone --quiet --shared --no-checkout "$source" "$dest"
        git -C "$dest" checkout --quiet --detach "$revision"
    fi
    git -C "$dest" rev-parse HEAD
}

checkout_submodules() {
    local source=$1 dest=$2 entry header path mode kind revision
    git -C "$dest" ls-tree -rz HEAD | while IFS= read -r -d '' entry; do
        header=${entry%%$'\t'*}
        path=${entry#*$'\t'}
        read -r mode kind revision <<< "$header"
        if [[ $kind == commit ]]; then
            if [[ ! -e "$source/$path/.git" ]]; then
                printf 'Required local submodule missing: %s\n' "$source/$path" >&2
                return 1
            fi
            printf '%s ' "${dest#"$workspace/"}"/"$path"
            checkout "$source/$path" "$revision" "$dest/$path"
            checkout_submodules "$source/$path" "$dest/$path"
        fi
    done
}

in_repo() (
    cd "$workspace/$1"
    shift
    "$@"
)

log_step() {
    local label=$1
    shift
    "$@" 2>&1 | tee "$logs/$label.log"
}

cat > "$acceptance/state_compat.py" <<'PY'
import sys
from pathlib import Path

from zeta_forge import conan_state as conan
from zeta_forge import native_transaction as tx
from zeta_forge.native_artifacts import RECEIPT_NAME, make_receipt, require_receipt

mode, root = sys.argv[1], Path(sys.argv[2])
stage = root / "stage"
prefix = root / "prefix"
state = prefix / tx.STATE_PATH
relative = "lib/phase3-probe"
published = prefix / relative
candidate = root / "candidate"
generation = root / "input-fingerprint"
lock = root / "conan.lock"

if mode == "write":
    stage.mkdir(parents=True)
    (stage / "probe").write_bytes(b"phase3")
    conan.write_json(stage / RECEIPT_NAME, make_receipt(
        stage, "nng", "release", sources=[], conan=None, references=[], dependencies={},
    ))
    conan.write_json(lock, {"version": "0.5", "requires": [], "build_requires": [],
                            "python_requires": [], "config_requires": []})
    identity = conan.store_generation(generation, lock, {
        "schema": 1, "input_fingerprint": generation.name, "generators": {},
        "graph": [], "lock_sha256": conan.file_digest(lock),
    })
    conan.write_json(generation / "active.json", {"schema": 1, "generation": identity})
    published.parent.mkdir(parents=True)
    published.write_bytes(b"before")
    candidate.write_bytes(b"interrupted")
    tx.prepare_transaction(prefix, state / "pending", {
        relative: (tx.file_identity(published), tx.file_identity(candidate), candidate),
    }, {"schema": 1, "transaction": "a" * 32, "owners": {}})
    tx.replace_entry(state / "pending" / "new" / relative, published)
else:
    require_receipt(stage, "nng", "release")
    found = conan.active_generation(generation)
    assert found is not None and found[1].read_bytes() == lock.read_bytes()
    assert published.read_bytes() == b"interrupted"
    tx.recover_transaction(prefix, state)
    assert published.read_bytes() == b"before" and not (state / "pending").exists()
    print(f"Receipt, lock and rollback passed: {root.name}")
PY

* 5. 启动性能对照和最终判定

在同一个环境中交替测量，减少冷热启动偏差。每个命令记录 12 组结果和中位数。
同时对照架构测试及台账，确认 native 注册、CMake 参数、状态 I/O 和 Rust manifest
命令各由一个实现负责。启动计时的判断依据是有无可复现的明显退化。

#+begin_src bash
python3 - "$acceptance/baseline" "$workspace/zeta_forge" "$acceptance/startup-prefix" <<'PY' | tee "$logs/startup.log"
import os
import statistics
import subprocess
import sys
import time
from pathlib import Path

baseline, candidate, prefix = map(Path, sys.argv[1:])
for command in (["metadata", "nng"], ["rebuild", "nng", "--dry-run"]):
    samples = {repo: [] for repo in (baseline, candidate)}
    for index in range(12):
        order = (baseline, candidate) if index % 2 == 0 else (candidate, baseline)
        for repo in order:
            env = dict(os.environ, ZETAX_ROOT=str(repo.parent), ZETA_INSTALL_PREFIX=str(prefix))
            start = time.perf_counter()
            subprocess.run([str(repo / "zbuild.py"), *command], cwd=repo, env=env,
                           stdout=subprocess.DEVNULL, stderr=subprocess.PIPE, check=True)
            samples[repo].append((time.perf_counter() - start) * 1000)
    print(" ".join(command))
    for label, repo in (("baseline", baseline), ("candidate", candidate)):
        print(f"{label}: median {statistics.median(samples[repo]):.1f} ms, "
              f"samples {[round(value, 1) for value in samples[repo]]}")
PY
#+end_src

核对全部命令日志、两项浏览器结果及启动对照后，把 =Phase 3 整体验收通过= 和验收
目录发到当前对话即可。若有失败或跳过，附对应步骤或日志文件。Android 真机安装及
VPN 连通性、实际交易服务均不在这份手册的验证范围内。
